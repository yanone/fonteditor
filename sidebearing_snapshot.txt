  uid=1_408 StaticText "# Visualize vertical distribution of top anchors
# Keywords: anchors, glyphs, layers
#
# Scans all glyphs and layers for anchors named 'top',
# collects their Y positions, computes statistics,
# and produces a two-panel matplotlib visualization.

import matplotlib.pyplot as plt
import numpy as np
from collections import defaultdict

font = Font()

# Collect all 'top' anchor Y positions with glyph names
anchor_data = defaultdict(list)  # glyph_name -> list of Y values
all_y = []
glyph_y_pairs = []  # (glyph_name, y) for labeling

for glyph in font.glyphs:
    if not glyph.layers:
        continue
    for layer in glyph.layers:
        if not layer.anchors:
            continue
        for anchor in layer.anchors:
            if anchor.name == "top":
                y = anchor.y
                anchor_data[glyph.name].append(y)
                all_y.append(y)
                glyph_y_pairs.append((glyph.name, y))

if not all_y:
    print("No anchors named 'top' found in any glyph.")
else:
    y_arr = np.array(all_y)
    mean_y = np.mean(y_arr)
    median_y = np.median(y_arr)
    std_y = np.std(y_arr)
    min_y = np.min(y_arr)
    max_y = np.max(y_arr)
    count = len(y_arr)
    
    # Identify outliers (more than 2 SD from mean)
    outliers = [(name, y) for name, y in glyph_y_pairs if abs(y - mean_y) > 2 * std_y]
    
    # Print statistics
    print(f"=== Top Anchor Y-Position Statistics ===")
    print(f"  Count:          {count}")
    print(f"  Mean:           {mean_y:.2f}")
    print(f"  Median:         {median_y:.2f}")
    print(f"  Std Deviation:  {std_y:.2f}")
    print(f"  Min:            {min_y:.2f}")
    print(f"  Max:            {max_y:.2f}")
    print(f"  Outliers (>{2*std_y:.2f} from mean):")
    if outliers:
        for name, y in sorted(outliers, key=lambda x: -abs(x[1] - mean_y)):
            dev = y - mean_y
            print(f"    {name}: y={y:.2f} (dev={dev:+.2f})")
    else:
        print("    None found")
    
    # Create two-panel visualization
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 6))
    fig.patch.set_facecolor('#1e1e2e')
    
    # --- Left panel: Histogram ---
    ax1.set_facecolor('#1e1e2e')
    ax1.hist(y_arr, bins=20, color='#89b4fa', edgecolor='#1e1e2e', alpha=0.85)
    ax1.axvline(mean_y, color='#f38ba8', linewidth=2, linestyle='-', label=f'Mean: {mean_y:.1f}')
    ax1.axvline(median_y, color='#f9e2af', linewidth=2, linestyle='--', label=f'Median: {median_y:.1f}')
    ax1.axvline(mean_y + std_y, color='#a6e3a1', linewidth=1, linestyle=':', label=f'±1 SD')
    ax1.axvline(mean_y - std_y, color='#a6e3a1', linewidth=1, linestyle=':')
    ax1.set_title('Distribution of Top Anchor Y Positions', color='white', fontsize=13, fontweight='bold')
    ax1.set_xlabel('Y Position (font units)', color='white')
    ax1.set_ylabel('Frequency', color='white')
    ax1.tick_params(colors='white')
    ax1.legend(facecolor='#313244', edgecolor='white', labelcolor='white')
    for spine in ax1.spines.values():
        spine.set_color('#585b70')
    
    # --- Right panel: Jittered dot plot ---
    ax2.set_facecolor('#1e1e2e')
    # Compute colors: green near mean, red far from mean (normalized)
    deviations = np.abs(y_arr - mean_y)
    max_dev = deviations.max() if deviations.max() > 0 else 1
    colors = plt.cm.RdYlGn_r(1.0 - deviations / max_dev)  # RdYlGn_r: red=0, green=1
    # Jitter X coordinates
    np.random.seed(42)
    jitter_x = np.random.normal(0, 0.05, count)
    
    ax2.scatter(jitter_x, y_arr, c=colors, alpha=0.7, s=30, edgecolors='white', linewidth=0.3)
    
    # Label outliers
    for name, y in outliers:
        ax2.annotate(name, (0, y), textcoords="offset points", xytext=(8, 4),
